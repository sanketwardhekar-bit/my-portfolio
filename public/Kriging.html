import numpy as np
from pandas import DataFrame, Series
from scipy.spatial.distance import pdist,squareform
inp = open('demo.txt', 'r').readlines()
inp = [i.strip().split() for i in inp]
inp = np.array(inp, dtype=np.float)
inp = DataFrame(inp, columns=['x', 'y', 'thk','por','perm','lperm','lpermp','lpermr'])


data = np.array(inp[['x', 'y','por']])
band = 500
dist = np.arange(0, 10500, band)

def svcal( data, dist1, band):
    pointdist = squareform(pdist(data[:, :2]))
    N = pointdist.shape[0]
    list1 = list()
    for i in range(N):
        for j in range(i+1, N):
            if(pointdist[i, j] >= dist1-band) and (pointdist[i, j] <= dist1+band):
                list1.append((data[i, 2] - data[j, 2])**2.0)
    return np.sum(list1)/(2.0*len(list1))

def semivar(data, dist, band):
    svlist1 = list()
    for dist1 in dist:
        svlist1.append(svcal(data, dist1, band))
    svlist1 = [[dist[i], svlist1[i]] for i in range(len(dist)) if svlist1[i] > 0]
    return np.array(svlist1).T

list1 = semivar(data, dist, band)

def sill(data, dist1, band):
    sillcal = np.var(data[:, 2])
    if dist1 == 0:
        return sillcal
    return sillcal - svcal(data, dist1, band)


def opt( fct, x, y, C0, parameterRange=None, meshSize=1000 ):
    if parameterRange == None:
        parameterRange = [x[1], x[-1]]
    mse = np.zeros(meshSize)
    a = np.linspace( parameterRange[0], parameterRange[1], meshSize )
    for i in range( meshSize):
        mse[i] = np.mean((y-fct(x, a[i], C0))**2.0)
    return a[mse.argmin()]

def spherical( dist1, a, C0 ):
    if type(dist1) == np.float64:
        if dist1 <= a:
            return C0*( 1.5*dist1/a - 0.5*(dist1/a)**3.0 )
        else:
            return C0
    else:
        a = np.ones( dist1.size ) * a
        C0 = np.ones( dist1.size ) * C0
        return map(spherical, dist1, a, C0)

def covmod(data, spherical, dist, band):
    list1 = semivar(data, dist, band)
    C0 = sill(data, dist[0], band)
    parameters = opt(spherical, list1[0], list1[1], C0)
    covfn = lambda dist1, a = parameters: C0 - spherical(dist1, a, C0)
    return covfn

    covfn = covmod(data, spherical, dist, band) #covariance function
    mean = np.mean(data[:, 2])              #mean value of variables
    dist2 = np.sqrt((data[:, 0]-prv[0])**2.0+(data[:, 1]-prv[1])**2.0) #distance between unknown and known points
    data = np.vstack((data.T, dist2)).T     #adding distances to data
    data = data[dist2.argsort()[:N]]        #sorting data by dist2 and taking N values
    covmat = covfn(data[:,3])
    covmat = np.matrix(covmat).T            #covariance matrix
    distmat = squareform(pdist(data[:, :2]))#matrix of dist2
    distmat = covfn(distmat.ravel())        #applying covariance spherical
    distmat = np.array(distmat)
    distmat = distmat.reshape(N,N)
    distmat = np.matrix(distmat)
    weights = np.linalg.inv(distmat)*covmat#multiplying inverse matrix of dist2 and covariance matrix
    weights = np.array(weights)
    residuals = data[:, 2] - mean           #calculating residuals
    predicted_values = np.dot(weights.T, residuals) + mean
    return float(predicted_values)
